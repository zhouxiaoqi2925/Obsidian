```java
// 服务发现客户端
@Service
@Slf4j
public class ServiceDiscovery {
    
    @Autowired
    private NamingService namingService;
    
    public List<Instance> getHealthyInstances(String serviceName) {
        try {
            // 获取所有健康实例
            List<Instance> instances = namingService.selectInstances(
                serviceName, 
                true  // healthy only
            );
            
            // 负载均衡：权重随机
            return weightedRandomSelect(instances);
        } catch (NacosException e) {
            log.error("服务发现失败: {}", serviceName, e);
            return Collections.emptyList();
        }
    }
    
    private List<Instance> weightedRandomSelect(List<Instance> instances) {
        int totalWeight = instances.stream()
            .mapToInt(Instance::getWeight)
            .sum();
        
        int random = ThreadLocalRandom.current().nextInt(totalWeight);
        int currentWeight = 0;
        
        for (Instance instance : instances) {
            currentWeight += instance.getWeight();
            if (random < currentWeight) {
                return Collections.singletonList(instance);
            }
        }
        return Collections.singletonList(instances.get(0));
    }
}
```

### 2.3 服务熔断降级

```java
// Sentinel 熔断配置
@Configuration
public class SentinelConfig {
    
    @Bean
    public SentinelRestTemplate sentinelRestTemplate() {
        return new SentinelRestTemplate();
    }
}

// 全局降级处理
@Component
public class FallbackHandler implements BlockExceptionHandler {
    
    @Override
    public handle(BlockException ex) {
        // 熔断触发
        if (ex instanceof DegradeException) {
            return Result.fail("系统繁忙，请稍后重试");
        }
        
        // 限流触发
        if (ex instanceof FlowException) {
            return Result.fail("请求过于频繁，请稍后重试");
        }
        
        // 热点限流
        if (ex instanceof ParamFlowException) {
            return Result.fail("访问过于频繁");
        }
        
        return Result.fail("服务暂不可用");
    }
}

// Sentinel 注解配置
@SentinelResource(
    value = "getUser",
    blockHandler = "getUserBlockHandler",
    fallback = "getUserFallback",
    exceptionsToTrace = {BusinessException.class}
)
public User getUser(Long userId) {
    return userRepository.findById(userId)
        .orElseThrow(() -> new UserNotFoundException(userId));
}

// 降级方法
public User getUserFallback(Long userId, Throwable throwable) {
    log.warn("获取用户降级: userId={}, error={}", userId, throwable.getMessage());
    return User.defaultUser(userId);
}
```

```java
// 分布式事务示例 - 订单创建
@GlobalTransactional(
    name = "create-order",
    rollbackFor = Exception.class,
    timeoutMills = 30000
)
public Order createOrder(OrderDTO orderDTO) {
    // 1. 创建订单（本地事务）
    Order order = orderRepository.save(Order.builder()
        .userId(orderDTO.getUserId())
        .amount(orderDTO.getAmount())
        .status(OrderStatus.PENDING)
        .build());
    
    // 2. 扣减库存（远程事务）
    inventoryClient.deductStock(orderDTO.getSkuId(), orderDTO.getQuantity());
    
    // 3. 扣减余额（远程事务）
    accountClient.deductBalance(orderDTO.getUserId(), orderDTO.getAmount());
    
    // 4. 发送消息
    messageClient.sendOrderCreated(order);
    
    return order;
}
```

### 2.5 消息队列 - RocketMQ事务消息

```java
// 事务消息实现
@Service
@Slf4j
public class OrderService {
    
    @Autowired
    private RocketMQTemplate rocketMQTemplate;
    
    public void createOrder(Order order) {
        // 发送半消息
        rocketMQTemplate.sendMessageInTransaction(
            "topic_order",
            MessageBuilder.withPayload(order)
                .setHeader("orderId", order.getId())
                .setHeader("userId", order.getUserId())
                .build(),
            order);  // transaction listener
    }
}

@Component
@Slf4j
public class OrderTransactionListener implements TransactionListener {
    
    @Autowired
    private OrderRepository orderRepository;
    
    @Override
    public LocalTransactionState executeLocalTransaction(Message msg, Object arg) {
        try {
            Order order = (Order) arg;
            
            // 本地事务：创建订单
            orderRepository.save(order);
            
            // 更新库存
            inventoryService.deductStock(order.getItems());
            
            return LocalTransactionState.COMMIT_MESSAGE;
        } catch (Exception e) {
            log.error("本地事务失败", e);
            return LocalTransactionState.ROLLBACK_MESSAGE;
        }
    }
    
    @Override
    public LocalTransactionState checkLocalTransaction(MessageExt msg) {
        String orderId = msg.getHeaders().get("orderId", String.class);
        
        // 检查订单状态
        Order order = orderRepository.findById(orderId);
        if (order == null) {
            return LocalTransactionState.ROLLBACK_MESSAGE;
        }
        
        if (order.getStatus() == OrderStatus.CONFIRMED) {
            return LocalTransactionState.COMMIT_MESSAGE;
        }
        
        return LocalTransactionState.UNKNOWN;
    }
}
```

## 3. 微服务拆分原则

### 3.1 拆分维度

| 维度 | 说明 | 示例 |
|------|------|------|
| 业务边界 | 高内聚、低耦合 | 订单服务、用户服务 |
| 数据边界 | 独立数据库 | 订单库、用户库 |
| 发布频率 | 不同发布周期 | 核心系统 vs 创新业务 |
| 团队边界 | 康威定律 | 独立团队负责独立服务 |
| 性能要求 | 不同SLA | 高频交易 vs 后台分析 |

### 3.2 服务粒度评估

```python
# 服务拆分评估模型
class ServiceSplitAnalyzer:
    def __init__(self, call_graph):
        self.graph = call_graph
    
    def analyze_coupling(self, candidate_service):
        """
        分析候选服务的耦合度
        """
        # 1. 圈内调用密度（高 = 适合拆分）
        internal_calls = self.graph.internal_calls(candidate_service)
        external_calls = self.graph.external_calls(candidate_service)
        
        internal_density = internal_calls / (internal_calls + external_calls)
        
        # 2. 数据一致性要求
        consistency_requirements = self.graph.data_consistency(candidate_service)
        
        # 3. 变更频率
        change_frequency = self.graph.change_rate(candidate_service)
        
        return {
            'internal_density': internal_density,
            'external_dependencies': external_calls,
            'consistency_score': consistency_requirements,
            'change_score': change_frequency,
            'recommendation': self._recommend_split(internal_density, external_calls)
        }
    
    def _recommend_split(self, density, external):
        if density > 0.7 and external < 5:
            return "HIGH_CONFIDENCE_SPLIT"
        elif density > 0.5:
            return "CONSIDER_SPLIT"
        else:
            return "DO_NOT_SPLIT"
```

## 4. 服务间通信

### 4.1 gRPC vs HTTP

| 特性 | gRPC | HTTP/REST |
|------|------|-----------|
| 协议 | HTTP/2 | HTTP/1.1-2 |
| 序列化 | Protobuf (二进制) | JSON |
| 性能 | 更高 | 一般 |
| 代码生成 | 强类型自动生成 | 手动/OpenAPI |
| 流式支持 | 双向流 | 单向 |
| 浏览器支持 | 需要gRPC-Web | 原生支持 |
| 适用场景 | 内部微服务 | 外部API |

```java
// OpenTelemetry 手动埋点
@Component
@Slf4j
public class TracingHelper {
    
    @Autowired
    private Tracer tracer;
    
    public <T> T executeWithTrace(String operationName, Supplier<T> operation) {
        Span span = tracer.spanBuilder(operationName).startSpan();
        
        try (Scope scope = span.makeCurrentScope()) {
            span.setAttribute("service.name", "user-service");
            span.setAttribute("operation.type", "database");
            
            T result = operation.get();
            
            span.setAttribute("result.status", "success");
            return result;
        } catch (Exception e) {
            span.setAttribute("result.status", "error");
            span.setAttribute("error.type", e.getClass().getSimpleName());
            span.setAttribute("error.message", e.getMessage());
            span.recordException(e);
            throw e;
        } finally {
            span.end();
        }
    }
    
    public void addEvent(String name, Map<String, Object> attributes) {
        Span currentSpan = Span.current();
        currentSpan.addEvent(name, toAttributes(attributes));
    }
}
```

## 7. 常见问题处理

### 7.1 服务雪崩

```
现象：
- 单个服务故障 → 级联失败 → 整个系统不可用

解决方案：
1. 熔断器（Hystrix/Sentinel）
2. 限流（Rate Limiter）
3. 舱壁隔离（Bulkhead）
4. 快速失败（Fail Fast）
5. 降级兜底（Fallback）
```

### 7.2 分布式事务一致性

| 方案 | 一致性 | 性能 | 复杂度 | 适用场景 |
|------|--------|------|--------|----------|
| 2PC | 强一致 | 低 | 高 | 跨库操作 |
| TCC | 最终一致 | 中 | 高 | 金融业务 |
| Saga | 最终一致 | 高 | 中 | 长流程 |
| 本地消息表 | 最终一致 | 高 | 低 | 普通业务 |
| 最大努力通知 | 最终一致 | 高 | 低 | 非核心业务 |
