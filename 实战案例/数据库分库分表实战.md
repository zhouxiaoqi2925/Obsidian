## 1. 分库分表架构

### 1.1 整体架构

```
┌────────────────────────────────────────────────────────────────┐
│                        ShardingSphere-JDBC                      │
│  ┌──────────────────────────────────────────────────────────┐   │
│  │                    Sharding Algorithm                    │   │
│  │  Hash分片 / Range分片 / 复合分片 / Hint分片               │   │
│  └──────────────────────────────────────────────────────────┘   │
└─────────────────────────┬──────────────────────────────────────┘
                          │
    ┌─────────────────────┼─────────────────────┐
    ▼                     ▼                     ▼
┌─────────┐          ┌─────────┐          ┌─────────┐
│ Shard 1 │          │ Shard 2 │          │ Shard N │
│ (DS_0) │          │ (DS_1) │          │ (DS_N) │
│ users_0 │          │ users_1 │          │ users_N │
│ orders_0│          │ orders_1│          │ orders_N│
└─────────┘          └─────────┘          └─────────┘
```

### 1.2 分片策略对比

| 策略 | 原理 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|----------|
| Hash分片 | `id % N` | 数据均匀 | 扩缩容麻烦 | 订单、交易流水 |
| Range分片 | `id BETWEEN` | 范围查询快 | 数据不均 | 冷热数据分离 |
| 时间分片 | `created_at` | 历史查询快 | 近期数据热点 | 日志、监控 |
| 复合分片 | 多字段组合 | 灵活 | 实现复杂 | 特殊业务需求 |

```yaml
            type: SNOWFLAKE
```

```java
// 复合分片算法
@Component
public class UserShardingAlgorithm implements StandardComplexShardingAlgorithm {
    
    private static final Map<String, Integer> REGION_DATABASE_MAP = Map.of(
        "CN", 0,
        "US", 1,
        "EU", 2
    );
    
    @Override
    public Collection<String> doSharding(
            Collection<String> availableTargetNames,
            ComplexShardingValue<Object> shardingValue) {
        
        String region = shardingValue.getValueMap()
            .get("region").toString();
        Long userId = Long.valueOf(
            shardingValue.getValueMap().get("user_id").toString()
        );
        
        int databaseIndex = REGION_DATABASE_MAP.getOrDefault(region, 0);
        int tableIndex = (int) (userId % 8);
        
        return Collections.singletonList(
            String.format("ds_%d.t_user_%d", databaseIndex, tableIndex)
        );
    }
}
```

## 3. 分布式ID生成

### 3.1 Snowflake 算法

```
┌────────────────────────────────────────────┐
│                 Snowflake ID               │
│  ┌──────┬──────────┬────────┬─────────┐   │
│  │符号位│  时间戳   │机器ID │ 序列号   │   │
│  │1bit │  41bit   │10bit  │ 12bit   │   │
│  └──────┴──────────┴────────┴─────────┘   │
│                                           │
│  时间戳：毫秒级，可支持69年                │
│  机器ID：支持1024台机器                    │
│  序列号：每毫秒支持4096个ID               │
└────────────────────────────────────────────┘
```

### 3.2 Snowflake 实现

```java
@Component
@Slf4j
public class SnowflakeIdGenerator {
    
    private final long twepoch = 1609459200000L; // 2021-01-01
    private final long workerIdBits = 10L;
    private final long maxWorkerId = ~(-1L << workerIdBits);
    private final long sequenceBits = 12L;
    private final long workerIdShift = sequenceBits;
    private final long timestampLeftShift = sequenceBits + workerIdBits;
    private final long sequenceMask = ~(-1L << sequenceBits);
    
    private final long workerId;
    private long sequence = 0L;
    private long lastTimestamp = -1L;
    
    public SnowflakeIdGenerator(@Value("${snowflake.worker-id}") long workerId) {
        if (workerId > maxWorkerId || workerId < 0) {
            throw new IllegalArgumentException("Worker ID 超出范围");
        }
        this.workerId = workerId;
    }
    
    public synchronized long nextId() {
        long timestamp = getCurrentTimestamp();
        
        if (timestamp < lastTimestamp) {
            throw new RuntimeException("时钟回拨，拒绝生成ID");
        }
        
        if (timestamp == lastTimestamp) {
            sequence = (sequence + 1) & sequenceMask;
            if (sequence == 0) {
                timestamp = waitNextMillis(timestamp);
            }
        } else {
            sequence = 0;
        }
        
        lastTimestamp = timestamp;
        
        return ((timestamp - twepoch) << timestampLeftShift)
            | (workerId << workerIdShift)
            | sequence;
    }
    
    private long getCurrentTimestamp() {
        return System.currentTimeMillis();
    }
    
    private long waitNextMillis(long timestamp) {
        while (timestamp <= lastTimestamp) {
            timestamp = getCurrentTimestamp();
        }
        return timestamp;
    }
    
    // 解析ID
    public Map<String, Long> parseId(long id) {
        long timestamp = (id >> timestampLeftShift) + twepoch;
        long worker = (id >> workerIdShift) & maxWorkerId;
        long seq = id & sequenceMask;
        return Map.of(
            "timestamp", timestamp,
            "workerId", worker,
            "sequence", seq
        );
    }
}
```

## 4. 跨分片查询

### 4.1 分页查询

```java
@Service
@Slf4j
public class ShardingPageQueryService {
    
    @Autowired
    private JdbcTemplate jdbcTemplate;
    
    @Autowired
    private ShardingSphereAutoConfiguration shardingSphere;
    
    /**
     * 跨分片分页查询
     * 需要广播查询所有分片，然后合并结果
     */
    public <T> Page<T> crossShardingPageQuery(
            String sql,
            Object[] params,
            int pageNum,
            int pageSize,
            RowMapper<T> rowMapper) {
        
        // 1. 获取所有实际数据节点
        List<String> actualDataNodes = getActualDataNodes();
        
        // 2. 并行查询所有分片
        List<Future<CrossShardResult<T>>> futures = new ArrayList<>();
        ExecutorService executor = Executors.newFixedThreadPool(
            Math.min(actualDataNodes.size(), 10)
        );
        
        for (String dataNode : actualDataNodes) {
            Future<CrossShardResult<T>> future = executor.submit(() -> {
                return executeQuery(dataNode, sql, params, pageNum, pageSize);
            });
            futures.add(future);
        }
        
        // 3. 合并结果
        List<T> allResults = new ArrayList<>();
        long totalCount = 0;
        
        for (Future<CrossShardResult<T>> future : futures) {
            CrossShardResult<T> result = future.get();
            allResults.addAll(result.getData());
            totalCount += result.getTotalCount();
        }
        
        executor.shutdown();
        
        // 4. 内存分页
        int startIndex = (pageNum - 1) * pageSize;
        int endIndex = Math.min(startIndex + pageSize, allResults.size());
        
        List<T> pageData = startIndex < allResults.size() 
            ? allResults.subList(startIndex, endIndex) 
            : Collections.emptyList();
        
        return new Page<>(pageData, totalCount, pageNum, pageSize);
    }
    
    private CrossShardResult<T> executeQuery(
            String dataNode,
            String sql,
            Object[] params,
            int pageNum,
            int pageSize) {
        
        // 执行实际查询
        List<T> results = jdbcTemplate.query(sql, params, rowMapper);
        return new CrossShardResult<>(results, results.size());
    }
}
```

### 4.2 跨分片聚合

```java
@Service
public class AggregationService {
    
    /**
     * 多分片聚合计算
     * 例如：计算所有订单的总金额
     */
    public BigDecimal calculateTotalAmount(Long userId) {
        String sql = "SELECT SUM(amount) FROM t_order WHERE user_id = ?";
        
        // 路由到特定分片
        List<String> actualTables = shardingRouter.route(
            "t_order", 
            ShardingValue.of("user_id", userId)
        );
        
        BigDecimal totalAmount = BigDecimal.ZERO;
        
        for (String table : actualTables) {
            BigDecimal tableAmount = jdbcTemplate.queryForObject(
                sql.replace("t_order", table),
                BigDecimal.class,
                userId
            );
            
            if (tableAmount != null) {
                totalAmount = totalAmount.add(tableAmount);
            }
        }
        
        return totalAmount;
    }
    
    /**
     * 分片结果排序合并
     */
    public <T extends Comparable> List<T> mergeSortResults(
            Map<String, List<T>> shardResults,
            int limit) {
        
        return shardResults.values().stream()
            .flatMap(List::stream)
            .sorted(Comparator.reverseOrder())
            .limit(limit)
            .collect(Collectors.toList());
    }
}
```

## 5. 数据迁移

### 5.1 双写迁移方案

```java
@Service
@Slf4j
public class DataMigrationService {
    
    private final AtomicBoolean dualWriteEnabled = new AtomicBoolean(false);
    private final AtomicBoolean readFromNew = new AtomicBoolean(false);
    
    /**
     * 阶段1: 双写（写旧库同时写新库）
     * 阶段2: 迁移历史数据
     * 阶段3: 切换读（先读新库，成功后写新库）
     */
    public enum MigrationPhase {
        WRITE_BOTH,      // 双写
        MIGRATING,       // 迁移数据
        READ_NEW,        // 切换读
        WRITE_NEW,       // 切换写
        CLEANUP          // 清理旧库
    }
    
    @Autowired
    private OldOrderRepository oldOrderRepository;
    
    @Autowired
    private NewOrderRepository newOrderRepository;
    
    /**
     * 写操作：支持双写
     */
    @Transactional
    public Order createOrder(Order order) {
        Order savedOrder = newOrderRepository.save(order);
        
        if (dualWriteEnabled.get()) {
            try {
                // 双写旧库
                oldOrderRepository.save(toOldOrder(order));
            } catch (Exception e) {
                log.error("双写旧库失败", e);
                // 告警通知
                alertService.send("双写失败", order.getId());
            }
        }
        
        return savedOrder;
    }
    
    /**
     * 读操作：根据阶段决定读哪个库
     */
    @Transactional(readOnly = true)
    public Optional<Order> getOrder(Long orderId) {
        if (readFromNew.get()) {
            return newOrderRepository.findById(orderId);
        } else {
            return oldOrderRepository.findById(orderId)
                .map(this::toNewOrder);
        }
    }
    
    /**
     * 历史数据迁移
     */
    public void migrateHistoricalData(long startId, long endId, int batchSize) {
        long offset = startId;
        
        while (offset < endId) {
            List<Order> orders = oldOrderRepository
                .findByIdBetween(offset, offset + batchSize);
            
            for (Order order : orders) {
                try {
                    newOrderRepository.save(toNewOrder(order));
                    log.info("迁移订单: {}", order.getId());
                } catch (DuplicateKeyException e) {
                    // 已迁移，跳过
                    log.debug("订单已迁移: {}", order.getId());
                }
            }
            
            offset += batchSize;
            
            // 每批次暂停，避免影响主库
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
    }
}
```

### 5.2 迁移校验

```java
@Service
@Slf4j
public class MigrationValidator {
    
    /**
     * 数据校验：对比新旧库记录数
     */
    public ValidationReport validateMigration(long startId, long endId) {
        ValidationReport report = new ValidationReport();
        
        // 1. 总记录数校验
        long oldCount = oldOrderRepository.countByIdBetween(startId, endId);
        long newCount = newOrderRepository.countByIdBetween(startId, endId);
        
        report.setOldCount(oldCount);
        report.setNewCount(newCount);
        report.setCountMatch(oldCount == newCount);
        
        // 2. 随机抽样校验
        List<Long> sampleIds = selectRandomSamples(startId, endId, 100);
        List<ValidationResult> sampleResults = new ArrayList<>();
        
        for (Long id : sampleIds) {
            Optional<OldOrder> oldOpt = oldOrderRepository.findById(id);
            Optional<NewOrder> newOpt = newOrderRepository.findById(id);
            
            if (oldOpt.isPresent() && newOpt.isPresent()) {
                boolean match = compareOrders(oldOpt.get(), newOpt.get());
                sampleResults.add(new ValidationResult(id, match, ""));
            } else if (oldOpt.isPresent() != newOpt.isPresent()) {
                sampleResults.add(new ValidationResult(id, false, "缺失记录"));
            }
        }
        
        report.setSampleResults(sampleResults);
        report.setSampleMatchRate(
            sampleResults.stream()
                .filter(ValidationResult::isMatch)
                .count() * 100.0 / sampleResults.size()
        );
        
        return report;
    }
    
    private boolean compareOrders(OldOrder oldOrder, NewOrder newOrder) {
        return Objects.equals(oldOrder.getAmount(), newOrder.getAmount())
            && Objects.equals(oldOrder.getStatus(), newOrder.getStatus())
            && Objects.equals(oldOrder.getUserId(), newOrder.getUserId());
    }
}
```

```java
@Service
@Slf4j
public class CompensationService {
    
    @Transactional
    public boolean executeWithCompensation(Tasks tasks) {
        try {
            tasks.execute();
            return true;
        } catch (Exception e) {
            // 记录补偿任务
            compensationTaskRepository.save(
                CompensationTask.builder()
                    .tasks(tasks)
                    .status(CompensationStatus.PENDING)
                    .retryCount(0)
                    .build()
            );
            throw e;
        }
    }
}
```
