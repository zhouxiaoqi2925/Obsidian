```c
// sds 头结构
struct __attribute__ ((__packed__)) sdshdr {
    int len;        // 已用长度
    int alloc;      // 总分配空间
    unsigned char flags;  // 类型标记
    char buf[];      // 柔性数组，实际字符串
};

// 相比C字符串的优势
typedef char *sds;

// O(1) 获取长度
static inline size_t sdslen(const sds s) {
    struct sdshdr *sh = (void*)(s-(sizeof(struct sdshdr)));
    return sh->len;
}

// 惰性空间释放，避免频繁realloc
sds sdsRemoveFreeSpace(sds s) {
    struct sdshdr *sh = (void*)(s-(sizeof(struct sdshdr)));
    sh->free = 0;
    realloc(size);  // 仅在必要时释放
}

// 空间预分配优化
sds sdsMakeRoomFor(sds s, size_t addlen) {
    struct sdshdr *sh = (void*)(s-(sizeof(struct sdshdr)));
    size_t len = sh->len + addlen;
    
    if (len < 1024*1024)
        len *= 2;  // 小于1MB时加倍
    else
        len += 1024*1024;  // 大于1MB时增加1MB
    
    return s;
}
```

```go
// 标准控制器模式实现
type Controller struct {
    name       string
    informer   cache.SharedIndexInformer
    queue      workqueue.RateLimitingInterface
    lister     Lister
    syncer     Syncer
}

func (c *Controller) Run(threadiness int, stopCh <-chan struct{}) {
    defer utilruntime.HandleCrash()
    
    // 启动 informer
    go c.informer.Run(stopCh)
    
    // 等待缓存同步
    if !cache.WaitForCacheSync(stopCh, c.informer.HasSynced) {
        return
    }
    
    // 启动 worker 线程
    for i := 0; i < threadiness; i++ {
        go wait.Until(c.worker, time.Second, stopCh)
    }
}

func (c *Controller) worker() {
    for {
        workItem, shutdown := c.queue.Get()
        if shutdown {
            return
        }
        
        // 核心：执行 reconciler
        err := c.reconcile(workItem)
        if err != nil {
            // 失败重试
            c.queue.AddRateLimited(workItem)
        }
        
        c.queue.Done(workItem)
    }
}

// 核心 reconcile 逻辑
func (c *Controller) reconcile(key string) error {
    // 1. 从缓存获取对象
    obj, exists, err := c.informer.GetIndexer().GetByKey(key)
    if err != nil {
        return err
    }
    
    // 2. 计算期望状态
    expected := c.calculateDesiredState(obj)
    
    // 3. 获取当前状态
    current := c.getCurrentState(obj)
    
    // 4. 比较差异
    if !reflect.DeepEqual(expected, current) {
        // 5. 应用变化
        return c.applyChanges(expected)
    }
    
    return nil
}
```

```java
public class KafkaProducer<K, V> {
    
    // 批量发送优化
    private final RecordAccumulator accumulator;
    
    public Future<RecordMetadata> send(ProducerRecord<K, V> record) {
        // 1. 序列化
        byte[] serializedKey = serializer.serialize(record.topic(), record.key());
        byte[] serializedValue = serializer.serialize(record.topic(), record.value());
        
        // 2. 计算分区
        int partition = partition(record, serializedKey, serializedValue, metadata);
        
        // 3. 追加到批次

            partition,
            timestamp,
            serializedKey,
            serializedValue,
            callbacks,
            maxBlockTimeout
        );
        
        // 4. 唤醒sender
        this.sender.wakeup();
        
        return batch.future();
    }
}
```

### 4.2 Leader 选举实现

```go
// etcd Raft 选举实现
type Raft struct {
    term      uint64      // 当前任期
    voteFor   uint64      // 投票给谁
    state     State       // Follower/Candidate/Leader
    log       *RaftLog    // 日志
    votes     map[uint64]bool  // 收到的投票
}

func (r *Raft) tickElection() {
    r.electionTimeout++
    
    if r.electionTimeout >= r.electionTick {
        r.electionTimeout = 0
        r.startElection()  // 发起选举
    }
}

func (r *Raft) startElection() {
    // 状态转为 Candidate
    r.state = StateCandidate
    r.term++
    r.voteFor = r.id  // 给自己投票
    r.votes = map[uint64]bool{r.id: true}
    
    // 获取最新日志信息
    lastLogIndex, lastLogTerm := r.log.lastInfo()
    
    // 广播 RequestVote
    for peer := range r.peers {
        go r.sendRequestVote(peer, &RequestVoteRequest{
            Term:         r.term,
            CandidateId:  r.id,
            LastLogIndex: lastLogIndex,
            LastLogTerm:  lastLogTerm,
        })
    }
}

func (r *Raft) handleRequestVote(req RequestVoteRequest) *RequestVoteResponse {
    // 1. 任期比较
    if req.Term > r.term {
        r.becomeFollower(req.Term, None)
    }
    
    // 2. 投票判断
    canVote := r.voteFor == None || r.voteFor == req.CandidateId
    
    // 3. 日志是否最新
    lastIndex, lastTerm := r.log.lastInfo()
    logUpToDate := req.LastTerm > lastTerm || 
                   (req.LastTerm == lastTerm && req.LastLogIndex >= lastIndex)
    
    if canVote && logUpToDate {
        r.voteFor = req.CandidateId
        return &RequestVoteResponse{
            Term:        r.term,
            VoteGranted: true,
        }
    }
    
    return &RequestVoteResponse{
        Term:        r.term,
        VoteGranted: false,
    }
}
```

### 4.3 日志复制

```go
// Leader 日志复制
func (r *Raft) replicateToPeer(peer uint64) {
    // 获取下一个要复制的日志索引
    nextIndex := r.nextIndex[peer]
    
    // 发送 AppendEntries
    entries := r.log.slice(nextIndex, r.log.Size())
    
    prevLogIndex, prevLogTerm := r.log.getInfo(nextIndex - 1)
    
    req := &AppendEntriesRequest{
        Term:         r.term,
        LeaderId:     r.id,
        PrevLogIndex: prevLogIndex,
        PrevLogTerm:  prevLogTerm,
        Entries:      entries,
        LeaderCommit: r.commitIndex,
    }
    
    response := r.sendAppendEntries(peer, req)
    
    if response.Success {
        // 复制成功，更新 matchIndex
        r.nextIndex[peer] = nextIndex + len(entries)
        r.matchIndex[peer] = r.nextIndex[peer] - 1
        
        // 尝试提交
        r.tryCommit()
    } else {
        // 复制失败，递减 nextIndex 重试
        r.nextIndex[peer]--
    }
}

// 提交判断：超过半数节点已复制
func (r *Raft) tryCommit() {
    for n := r.commitIndex + 1; n <= r.log.Size(); n++ {
        if r.log.getTerm(n) == r.term {
            count := 1  // Leader自身
            
            for peer := range r.peers {
                if r.matchIndex[peer] >= n {
                    count++
                }
            }
            
            // 超过半数
            if count > len(r.peers)/2 {
                r.commitIndex = n
                break
            }
        }
    }
}
```

```javascript
// React Fiber 协调过程
function performUnitOfWork(fiber) {
    // 1. 创建子 Fiber
    const child = beginWork(fiber);
    
    if (child) {
        // 有子节点，优先处理子节点
        return child;
    }
    
    // 2. 没有子节点，完成当前 Fiber
    while (fiber) {
        completeWork(fiber);
        
        if (fiber.sibling) {
            // 有兄弟节点，返回兄弟节点
            return fiber.sibling;
        }
        
        // 3. 回溯到父节点
        fiber = fiber.return;
    }
}

    while (nextUnitOfWork && deadline.timeRemaining() > 0) {
```

```go
// Go Channel 底层实现
type hchan struct {
    qcount   uint           // 队列中的元素数
    dataqsiz uint           // 环形队列大小
    buf      unsafe.Pointer // 环形队列缓冲区
    elemsize uint16         // 元素大小
    closed   uint32         // 是否关闭
    elemtype *_type         // 元素类型
    sendx    uint           // 发送索引
    recvx    uint           // 接收索引
    recvq    waitq          // 接收等待队列 (阻塞的G)
    sendq    waitq          // 发送等待队列 (阻塞的G)
    lock     mutex          // 锁
}

type waitq struct {
    first *g
    last  *g
}

// 发送操作
func chansend(c *hchan, ep unsafe.Pointer, block bool, callerpc uintptr) bool {
    // 加锁
    lock(&c.lock)
    
    // 接收队列有等待者，直接复制
    if c.recvq.first != nil {
        gp := c.recvq.first
        c.recvq.first = gp.schedlink
        if c.recvq.first == nil {
            c.recvq.last = nil
        }
        unlock(&c.lock)
        
        // 唤醒接收者
        goready(gp, 3)
        return true
    }
    
    // 队列未满，加入队列
    if c.qcount < c.dataqsiz {
        c.buf[c.sendx] = ep
        c.sendx++
        if c.sendx == c.dataqsiz {
            c.sendx = 0
        }
        c.qcount++
        unlock(&c.lock)
        return true
    }
    
    if !block {
        unlock(&c.lock)
        return false
    }
    
    // 阻塞，加入发送队列
    gp := getg()
    gp.elem = ep
    gp.schedlink = nil
    c.sendq.last = gp
    c.sendq.last = gp
    
    // 切换到其他G
    gopark(...)
    return true
}
```

```go
// Context 取消传播
type Context interface {
    Deadline() (deadline time.Time, ok bool)
    Done() <-chan struct{}
    Err() error
    Value(key any) any
}

type cancelCtx struct {
    mu    sync.Mutex
    done  chan struct{}
    children map[canceller]struct{}
    err error
}

func (c *cancelCtx) Done() <-chan struct{} {
    return c.done
}

func (c *cancelCtx) cancel(removeFromParent bool, err error) {
    c.mu.Lock()
    if c.err == nil {
        c.err = err
        close(c.done)  // 关闭通知通道
    }
    
    // 取消所有子 Context
    for child := range c.children {
        child.cancel(false, err)
    }
    
    c.children = nil
    c.mu.Unlock()
}
```

### 10.2 Transformer 实现

```python
# 简化的 Transformer 实现
import torch
import torch.nn as nn
import math

class MultiHeadAttention(nn.Module):
    def __init__(self, d_model, num_heads):
        super().__init__()
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_k = d_model // num_heads
        
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.W_o = nn.Linear(d_model, d_model)
    
    def forward(self, x, mask=None):
        batch_size = x.size(0)
        
        # 线性变换
        Q = self.W_q(x).view(batch_size, -1, self.num_heads, self.d_k)
        K = self.W_k(x).view(batch_size, -1, self.num_heads, self.d_k)
        V = self.W_v(x).view(batch_size, -1, self.num_heads, self.d_k)
        
        # 转置以便计算
        Q = Q.transpose(1, 2)
        K = K.transpose(1, 2)
        V = V.transpose(1, 2)
        
        # Scaled Dot-Product Attention
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        
        if mask is not None:
            scores = scores.masked_fill(mask == 0, -1e9)
        
        attention = torch.softmax(scores, dim=-1)
        context = torch.matmul(attention, V)
        
        # 合并多头
        context = context.transpose(1, 2).contiguous()
        context = context.view(batch_size, -1, self.d_model)
        
        return self.W_o(context)

class TransformerBlock(nn.Module):
    def __init__(self, d_model, num_heads, d_ff):
        super().__init__()
        self.attention = MultiHeadAttention(d_model, num_heads)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_ff),
            nn.ReLU(),
            nn.Linear(d_ff, d_model)
        )
    
    def forward(self, x, mask=None):
        # Self-Attention + 残差
        attn_out = self.attention(x, mask)
        x = self.norm1(x + attn_out)
        
        # FFN + 残差
        ffn_out = self.ffn(x)
        x = self.norm2(x + ffn_out)
        
        return x
```
