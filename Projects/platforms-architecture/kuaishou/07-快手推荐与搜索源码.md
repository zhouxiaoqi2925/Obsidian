## 1. 快手 vs 抖音：去中心化分发的核心差异

### 1.1 算法哲学差异

| 维度 | 快手 | 抖音 |
|------|------|------|
| 核心哲学 | 普惠流量、关注关系 | 爆款逻辑、中心化分发 |
| 流量分配 | 基尼系数约束，新人曝光保障 | 头部聚集，爆款放大 |
| 关注权重 | 高 (关注 tab 占主要流量) | 弱化关注，强化推荐 |
| 用户意图 | 关系链驱动 | 算法推荐驱动 |
| 内容池 | UGC 真实记录 | 精致化运营 |
| 典型指标 | 关注转化率 | 完播率 |

### 1.2 基尼系数约束的流量分配（**[公开演讲]** 宿华多次提到）

```python
# 来源: 基于快手 CEO 宿华公开演讲 + 快手技术博客
# 快手 "公平普惠" 算法核心: 基尼系数约束
class FairnessAllocator:
    """快手流量分配器
    核心: 用基尼系数约束流量分配，避免头部过度集中
    """
    def __init__(self):
        # 基尼系数阈值 (越低越平均)
        self.gini_threshold = 0.6
        # 视频曝光基尼
        self.video_gini = 0.0
        # 创作者曝光基尼
        self.author_gini = 0.0

    def gini_coefficient(self, values):
        """计算基尼系数"""
        sorted_v = sorted(values)
        n = len(sorted_v)
        cum = sum((i + 1) * v for i, v in enumerate(sorted_v))
        return (2 * cum) / (n * sum(sorted_v)) - (n + 1) / n

    def adjust_allocation(self, candidates):
        """调整候选集的曝光权重
        candidates: [(video_id, score, base_exposure), ...]
        """
        # 1. 按创作者聚合曝光
        author_exposure = {}
        for vid, score, base_exp in candidates:
            author = self.get_author(vid)
            author_exposure[author] = author_exposure.get(author, 0) + base_exp

        # 2. 计算当前基尼
        current_gini = self.gini_coefficient(list(author_exposure.values()))

        # 3. 如果基尼过高，降权头部创作者
        if current_gini > self.gini_threshold:
            # 对头部 10% 创作者降权
            sorted_authors = sorted(
                author_exposure.items(),
                key=lambda x: -x[1]
            )
            top_k = max(1, len(sorted_authors) // 10)
            top_authors = set(a for a, _ in sorted_authors[:top_k])

            adjusted = []
            for vid, score, base_exp in candidates:
                author = self.get_author(vid)
                if author in top_authors:
                    # 降权 30%
                    adjusted.append((vid, score * 0.7, base_exp))
                else:
                    adjusted.append((vid, score, base_exp))
            return adjusted

        return candidates
```

```python
class ColdStartPool:
    """快手新人冷启动流量池
    - 新作品前 100 次曝光，从独立流量池分配
    - 看初始互动数据决定是否进入主流量池
    """
    def __init__(self):
        self.pool_size = 100  # 100 个独立曝光位
        self.initial_ctr_threshold = 0.1  # CTR > 10% 进入主池
        self.initial_vtr_threshold = 0.3  # 完播率 > 30%

    def check_promotion(self, video):
        """判断是否升级到主流量池"""
        ctr = video.exposed_click / max(video.exposed_show, 1)
        vtr = video.finished_view / max(video.exposed_show, 1)
        like_rate = video.likes / max(video.exposed_show, 1)
        comment_rate = video.comments / max(video.exposed_show, 1)

        # 综合分
        score = (ctr * 0.4 +
                 vtr * 0.3 +
                 like_rate * 0.2 +
                 comment_rate * 0.1)

        if score > self.initial_ctr_threshold:
            return True  # 进入主流量池
        return False

    def inject_to_main(self, video):
        """注入主流量池"""
        # 给一个基础曝光，再由精排决定后续
        base_exposure = 1000
        video.exposure_budget = base_exposure
        video.in_main_pool = True
```

### 2.1 双塔召回模型（**[公开演讲]** + 业界典型）

```python
# 来源: 基于快手技术博客 + 业界典型双塔架构
import torch
import torch.nn as nn


class UserTower(nn.Module):
    """用户塔
    输入: 用户行为序列 + 用户画像
    输出: user_embedding [B, D]
    """
    def __init__(self, num_items, num_tags, embed_dim=128, hidden=256):
        super().__init__()
        # Item embedding
        self.item_emb = nn.Embedding(num_items, embed_dim)
        # 行为序列 Transformer
        self.behavior_encoder = nn.TransformerEncoder(
            nn.TransformerEncoderLayer(
                d_model=embed_dim,
                nhead=4,
                dim_feedforward=hidden,
                batch_first=True,
            ),
            num_layers=2,
        )
        # 用户画像
        self.user_mlp = nn.Sequential(
            nn.Linear(num_tags, hidden),
            nn.ReLU(),
            nn.Linear(hidden, embed_dim),
        )
        # 融合 + 输出
        self.fusion = nn.Sequential(
            nn.Linear(embed_dim * 2, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim),
        )

    def forward(self, behavior_ids, user_tags):
        # 行为序列编码
        seq = self.item_emb(behavior_ids)  # [B, L, D]
        seq_feat = self.behavior_encoder(seq).mean(dim=1)  # [B, D]
        # 用户画像
        user_feat = self.user_mlp(user_tags)  # [B, D]
        # 融合
        combined = torch.cat([seq_feat, user_feat], dim=-1)
        return self.fusion(combined)


class ItemTower(nn.Module):
    """视频塔
    输入: 视频特征 (ID + 内容 + 创作者)
    输出: item_embedding [B, D]
    """
    def __init__(self, num_items, num_authors, embed_dim=128):
        super().__init__()
        self.item_emb = nn.Embedding(num_items, embed_dim)
        self.author_emb = nn.Embedding(num_authors, embed_dim)
        # 内容特征 (封面图 embedding + 文本 embedding)
        self.content_mlp = nn.Sequential(
            nn.Linear(512, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim),
        )
        self.fusion = nn.Linear(embed_dim * 3, embed_dim)

    def forward(self, item_ids, author_ids, content_feat):
        item = self.item_emb(item_ids)
        author = self.author_emb(author_ids)
        content = self.content_mlp(content_feat)
        combined = torch.cat([item, author, content], dim=-1)
        return self.fusion(combined)


class TwoTowerRecall(nn.Module):
    """双塔召回主模型"""
    def __init__(self, num_items, num_authors, num_tags):
        super().__init__()
        self.user_tower = UserTower(num_items, num_tags)
        self.item_tower = ItemTower(num_items, num_authors)
        self.temperature = 0.07

    def forward(self, behavior_ids, user_tags, item_ids, author_ids, content):
        user_emb = self.user_tower(behavior_ids, user_tags)
        item_emb = self.item_tower(item_ids, author_ids, content)

        # 归一化 (用于 cosine 相似度)
        user_emb = nn.functional.normalize(user_emb, dim=-1)
        item_emb = nn.functional.normalize(item_emb, dim=-1)

        # 余弦相似度
        logits = user_emb @ item_emb.T / self.temperature
        return logits

    def recall(self, user_emb, item_db, top_k=1000):
        """线上召回"""
        user_emb = nn.functional.normalize(user_emb, dim=-1)
        item_db = nn.functional.normalize(item_db, dim=-1)
        scores = item_db @ user_emb  # [num_items]
        top_scores, top_idx = torch.topk(scores, top_k)
        return top_idx, top_scores
```

### 2.2 行为序列建模：Transformer + 长序列优化

```python
# 来源: 基于快手 SIGIR 论文 + 业界典型
class LongSequenceUserEncoder(nn.Module):
    """长序列用户行为编码器
    快手用户行为序列可能长达 10000+，需要优化
    """
    def __init__(self, embed_dim=128, max_seq=10000):
        super().__init__()
        self.item_emb = nn.Embedding(20000000, embed_dim)

        # SIM (Search-based Interest Model) 取代全 attention
        self.sim_top_k = 200  # 只和 top-200 相关 item 做 attention
        self.sim_proj = nn.Linear(embed_dim, embed_dim)

    def forward(self, behavior_ids):
        """behavior_ids: [B, L] L 可以很长"""
        # General Search Interest: 聚合所有
        all_emb = self.item_emb(behavior_ids)  # [B, L, D]
        gsi = all_emb.mean(dim=1)  # [B, D]

        # Search-based Interest: 用 GSI 做 query，搜 top-k 相关 item
        # 实际生产中用 ANN 索引 (Faiss)
        sim_query = self.sim_proj(gsi)  # [B, D]

        # 找到 top-k 相关 item
        # scores: [B, L] = item_emb @ sim_query
        scores = (all_emb * sim_query.unsqueeze(1)).sum(dim=-1)
        top_k_vals, top_k_idx = torch.topk(scores, self.sim_top_k, dim=-1)

        # 只对 top-k item 做 attention
        top_k_emb = torch.gather(
            all_emb, 1,
            top_k_idx.unsqueeze(-1).expand(-1, -1, all_emb.shape[-1])
        )
        # 注意力汇聚
        attn_weights = torch.softmax(top_k_vals, dim=-1)
        esi = (top_k_emb * attn_weights.unsqueeze(-1)).sum(dim=1)

        # 融合 GSI + ESI
        return torch.cat([gsi, esi], dim=-1)
```

```python
# 来源: 基于快手技术博客 + 业界典型
class CoarseRanker(nn.Module):
    """粗排模型
    - 输入: 候选 video + user 上下文 (~3000 个)
    - 输出: 粗排分数 (~100 个)
    - 约束: 延迟 < 10ms
    """
    def __init__(self, num_items, embed_dim=64):
        super().__init__()
        self.item_emb = nn.Embedding(num_items, embed_dim)
        self.user_emb = nn.Linear(64, embed_dim)
        # 简单 MLP，避免过重
        self.mlp = nn.Sequential(
            nn.Linear(embed_dim * 2 + 10, 64),  # +10 为交叉特征
            nn.ReLU(),
            nn.Linear(64, 1),
        )

    def forward(self, user_feat, item_ids, cross_features):
        item = self.item_emb(item_ids)
        user = self.user_emb(user_feat)
        x = torch.cat([user, item, cross_features], dim=-1)
        return self.mlp(x).squeeze(-1)
```

## 4. 精排层：多目标深度模型

### 4.1 DIN (Deep Interest Network) - 快手深度兴趣网络

```python
# 来源: DIN 论文 + 快手定制改造 (业界典型 DIN 在快手有大规模应用)
class DINAttention(nn.Module):
    """DIN 的核心: 局部激活单元
    用候选 item 作为 query，去激活历史行为
    """
    def __init__(self, embed_dim=128):
        super().__init__()
        # 三个权重网络
        self.W = nn.Sequential(
            nn.Linear(embed_dim * 4, 64),
            nn.ReLU(),
            nn.Linear(64, 1),
        )

    def forward(self, candidate, history, mask=None):
        """candidate: [B, D]
           history: [B, L, D]
           mask: [B, L] (历史长度 mask)
        """
        # 复制 candidate 拼接 history
        L = history.shape[1]
        cand_expand = candidate.unsqueeze(1).expand(-1, L, -1)  # [B, L, D]
        # 4 个特征拼接: cand, hist, cand*hist, cand-hist
        attn_input = torch.cat([
            cand_expand,
            history,
            cand_expand * history,
            cand_expand - history,
        ], dim=-1)  # [B, L, 4D]

        # 注意力分数
        scores = self.W(attn_input).squeeze(-1)  # [B, L]
        if mask is not None:
            scores = scores.masked_fill(mask == 0, -1e9)

        # Softmax
        attn_weights = torch.softmax(scores, dim=-1)  # [B, L]
        # 加权求和
        output = (attn_weights.unsqueeze(-1) * history).sum(dim=1)  # [B, D]
        return output


class DINRanker(nn.Module):
    """DIN 精排模型"""
    def __init__(self, num_items, embed_dim=128):
        super().__init__()
        self.item_emb = nn.Embedding(num_items, embed_dim)
        self.user_profile_mlp = nn.Linear(64, embed_dim)
        self.din_attn = DINAttention(embed_dim)
        self.cross_mlp = nn.Sequential(
            nn.Linear(embed_dim * 3, 256),
            nn.ReLU(),
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Linear(128, 1),
        )

    def forward(self, user_profile, history_ids, history_mask, candidate_id):
        # 用户画像
        user_feat = self.user_profile_mlp(user_profile)
        # 历史行为
        hist_emb = self.item_emb(history_ids)
        # DIN 注意力
        cand_emb = self.item_emb(candidate_id)
        din_feat = self.din_attn(cand_emb, hist_emb, history_mask)
        # 交叉
        cross = torch.cat([user_feat, din_feat, cand_emb], dim=-1)
        ctr = torch.sigmoid(self.cross_mlp(cross))
        return ctr
```

### 4.2 DIEN (Deep Interest Evolution Network) - 时序建模

```python
# 来源: DIEN 论文 + 快手落地改造
class InterestEvolutionLayer(nn.Module):
    """DIEN 的兴趣演化层
    用 GRU + AUGRU 建模兴趣随时间演化
    """
    def __init__(self, embed_dim=128):
        super().__init__()
        # 普通 GRU 提取兴趣
        self.gru = nn.GRU(embed_dim, embed_dim, batch_first=True)
        # AUGRU: Attention Update Gate
        self.attention_mlp = nn.Sequential(
            nn.Linear(embed_dim * 2, 64),
            nn.ReLU(),
            nn.Linear(64, 1),
        )

    def augru_step(self, h, x, score):
        """AUGRU 单步
        h: [B, D] 上一时刻隐藏状态
        x: [B, D] 当前输入
        score: [B, 1] 注意力分数
        """
        # 标准 GRU gate
        r = torch.sigmoid(self.W_r(x) + self.U_r(h))
        z = torch.sigmoid(self.W_z(x) + self.U_z(h))
        h_tilde = torch.tanh(self.W_h(x) + r * self.U_h(h))
        # AUGRU: 更新门乘以注意力分数
        z = z * score  # 关键差异!
        h_new = (1 - z) * h + z * h_tilde
        return h_new

    def forward(self, history_emb, target_emb):
        """history_emb: [B, L, D]
           target_emb: [B, D]
        """
        # 1. 用 GRU 提取兴趣序列
        outputs, _ = self.gru(history_emb)  # [B, L, D]

        # 2. 计算每步兴趣与 target 的相关性
        target_expand = target_emb.unsqueeze(1).expand(-1, history_emb.shape[1], -1)
        attn_input = torch.cat([outputs, target_expand], dim=-1)
        attn_scores = self.attention_mlp(attn_input).squeeze(-1)  # [B, L]
        attn_scores = torch.softmax(attn_scores, dim=-1).unsqueeze(-1)  # [B, L, 1]

        # 3. AUGRU 演化
        h = outputs[:, 0]  # 初始隐藏
        for t in range(1, history_emb.shape[1]):
            score = attn_scores[:, t]
            h = self.augru_step(h, outputs[:, t], score)

        return h  # [B, D]
```

### 4.3 多目标学习：MMoE / PLE

```python
# 来源: MMoE 论文 + 业界在快手的应用
class MMoE(nn.Module):
    """MMoE: Multi-gate Mixture of Experts
    快手精排的核心: 同时预测 CTR / 完播率 / 点赞 / 评论 / 转发 / 收藏 / 关注
    """
    def __init__(self, num_tasks=5, num_experts=8, input_dim=256):
        super().__init__()
        # 专家网络
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(input_dim, 256),
                nn.ReLU(),
                nn.Linear(256, 256),
                nn.ReLU(),
            )
            for _ in range(num_experts)
        ])
        # 每个任务的门控
        self.gates = nn.ModuleList([
            nn.Linear(input_dim, num_experts)
            for _ in range(num_tasks)
        ])
        # 任务塔
        self.task_towers = nn.ModuleList([
            nn.Sequential(
                nn.Linear(256, 64),
                nn.ReLU(),
                nn.Linear(64, 1),
            )
            for _ in range(num_tasks)
        ])

    def forward(self, x):
        """x: [B, input_dim]"""
        # 1. 所有专家并行
        expert_outputs = torch.stack(
            [expert(x) for expert in self.experts],
            dim=1
        )  # [B, num_experts, 256]

        # 2. 每个任务独立 gate
        task_outputs = []
        for i in range(len(self.gates)):
            gate_scores = torch.softmax(self.gates[i](x), dim=-1)  # [B, num_experts]
            # 加权组合
            task_feat = (gate_scores.unsqueeze(-1) * expert_outputs).sum(dim=1)  # [B, 256]
            # 任务塔
            task_out = self.task_towers[i](task_feat)  # [B, 1]
            task_outputs.append(task_out)

        return task_outputs  # [(B,1), (B,1), ...]


class KuaishouMultiTaskModel(nn.Module):
    """快手多任务精排模型
    目标: [CTR, 完播率, 点赞, 评论, 收藏, 关注, 转发, 主页停留时长]
    """
    def __init__(self, num_items, num_users, embed_dim=128):
        super().__init__()
        self.item_emb = nn.Embedding(num_items, embed_dim)
        self.user_emb = nn.Embedding(num_users, embed_dim)
        self.din_attn = DINAttention(embed_dim)

        # 共享底层
        self.shared_bottom = nn.Sequential(
            nn.Linear(embed_dim * 4, 512),
            nn.ReLU(),
        )
        # MMoE 多目标
        self.mmoe = MMoE(
            num_tasks=8,
            num_experts=8,
            input_dim=512,
        )

    def forward(self, user_id, history, candidate_id, context):
        user = self.user_emb(user_id)
        cand = self.item_emb(candidate_id)
        hist_attn = self.din_attn(cand, history)

        combined = torch.cat([user, hist_attn, cand, context], dim=-1)
        shared = self.shared_bottom(combined)

        # 8 个目标
        ctr, vtr, like, comment, fav, follow, share, dwell = self.mmoe(shared)
        return {
            'ctr': ctr,
            'vtr': vtr,
            'like': like,
            'comment': comment,
            'fav': fav,
            'follow': follow,
            'share': share,
            'dwell': dwell,
        }
```

```python
# 来源: 业界典型重排架构
class Reranker:
    """重排层: 解决精排后的多样性、频次控制、上下文问题"""
    def __init__(self):
        self.diversity_threshold = 0.7  # 多样性阈值
        self.max_same_author = 2  # 同创作者最多 2 个
        self.max_same_tag = 3

    def rerank(self, candidates, user_history):
        """candidates: [(vid, score, item_info), ...] 精排后"""
        # 1. 多样性打散 (规则)
        # - 同创作者不能超过 max_same_author
        # - 同标签不能超过 max_same_tag
        result = []
        author_count = {}
        tag_count = {}

        # 按分数排序
        candidates = sorted(candidates, key=lambda x: -x[1])

        for vid, score, info in candidates:
            author = info['author_id']
            tags = info['tags']

            if author_count.get(author, 0) >= self.max_same_author:
                continue  # 跳过

            tag_overflow = False
            for t in tags:
                if tag_count.get(t, 0) >= self.max_same_tag:
                    tag_overflow = True
                    break
            if tag_overflow:
                continue

            result.append((vid, score, info))
            author_count[author] = author_count.get(author, 0) + 1
            for t in tags:
                tag_count[t] = tag_count.get(t, 0) + 1

            if len(result) >= 6:  # 一次只出 6 个
                break

        return result
```

```python
class VideoSearchRanker(nn.Module):
    """快手视频搜索精排
    输入: query + video title + video ocr + video tag
    输出: 相关性分数
    """
    def __init__(self, bert_model='hfl/chinese-roberta-wwm-ext'):
        super().__init__()
        self.bert = AutoModel.from_pretrained(bert_model)
        self.score_mlp = nn.Sequential(
            nn.Linear(768 * 3, 256),
            nn.ReLU(),
            nn.Linear(256, 1),
        )

    def encode_pair(self, query, doc):
        """query-doc 拼接输入 BERT"""
        text = f"[CLS] {query} [SEP] {doc} [SEP]"
        features = self.bert(text)
        return features.last_hidden_state[:, 0]  # [CLS]

    def forward(self, query, title, ocr_text, tag):
        # query-title 匹配
        qt = self.encode_pair(query, title)
        # query-OCR 匹配
        qo = self.encode_pair(query, ocr_text)
        # query-tag 匹配
        qtag = self.encode_pair(query, tag)

        # 拼接 + 打分
        combined = torch.cat([qt, qo, qtag], dim=-1)
        score = torch.sigmoid(self.score_mlp(combined))
        return score
```

### 7.1 Pasa 架构

```python
# 来源: KwaiYys/Pasa - pasa/model.py 简化
"""Pasa: LLM-based Personalized Assistant for Search ranking
核心思想: 用 LLM 替代/辅助精排模型
"""
import torch
import torch.nn as nn
from transformers import AutoModelForCausalLM, AutoTokenizer


class PasaRanker(nn.Module):
    """Pasa 模型: 用 LLM 做个性化搜索排序"""
    def __init__(self, llm_path='Qwen/Qwen-7B'):
        super().__init__()
        self.tokenizer = AutoTokenizer.from_pretrained(llm_path)
        # 加载冻结的 LLM
        self.llm = AutoModelForCausalLM.from_pretrained(
            llm_path,
            torch_dtype=torch.float16,
            device_map='auto',
        )
        # LoRA 微调
        from peft import LoraConfig, get_peft_model
        lora_config = LoraConfig(
            r=16,
            lora_alpha=32,
            target_modules=['q_proj', 'v_proj'],
            lora_dropout=0.05,
            bias='none',
        )
        self.llm = get_peft_model(self.llm, lora_config)

        # 排序头 (基于 LLM 隐藏状态)
        self.rank_head = nn.Sequential(
            nn.Linear(4096, 1024),
            nn.ReLU(),
            nn.Linear(1024, 1),
        )

    def build_prompt(self, user_history, query, candidates):
        """构造 prompt: 用户历史 + query + 候选列表"""
        prompt = f"""你是一个个性化推荐助手。请根据用户历史和当前 query，对候选视频排序。

用户最近观看:
{chr(10).join(f'- {v}' for v in user_history)}

当前搜索: {query}

候选视频:
"""
        for i, cand in enumerate(candidates):
            prompt += f"{i+1}. {cand['title']} | 标签: {cand['tags']} | 时长: {cand['duration']}s\n"

        prompt += """
请按相关性和用户兴趣从高到低输出编号 (例: 3,1,2,5,4):"""
        return prompt

    def forward(self, user_history, query, candidates):
        prompt = self.build_prompt(user_history, query, candidates)
        inputs = self.tokenizer(
            prompt,
            return_tensors='pt',
            truncation=True,
            max_length=4096,
        ).to(self.llm.device)

        # LLM 推理
        with torch.amp.autocast('cuda'):
            outputs = self.llm(
                **inputs,
                output_hidden_states=True,
                return_dict=True,
            )

        # 取最后一个 token 的隐藏状态
        hidden = outputs.hidden_states[-1][:, -1, :]  # [1, 4096]
        score = self.rank_head(hidden)
        return score

    def rerank(self, user_history, query, candidates):
        """Pasa 重排序"""
        scores = []
        for cand in candidates:
            score = self.forward(user_history, query, [cand])
            scores.append(score.item())
        # 按分数排序
        ranked = sorted(
            zip(candidates, scores),
            key=lambda x: -x[1]
        )
        return [c for c, s in ranked]
```

```python
        # 未点击的为负样本
        data = []
        for log in self.search_logs:
            user_hist = log['user_history'][-50:]  # 最近 50 个
            query = log['query']
            candidates = log['candidates']
            # 排序标签: 按点击位置降序
            labels = sorted(
                candidates,
                key=lambda x: -x['click_position']
            )
            data.append({
                'history': user_hist,
                'query': query,
                'candidates': candidates,
                'labels': labels,
            })
        return data

    def train_step(self, batch):
        """训练一步"""
        prompt = self.model.build_prompt(
            batch['history'],
            batch['query'],
            batch['candidates']
        )
        target = self.model.build_target(batch['labels'])

        inputs = self.model.tokenizer(
            prompt,
            return_tensors='pt',
        ).to(self.model.llm.device)

        target_ids = self.model.tokenizer(
            target,
            return_tensors='pt',
        ).input_ids.to(self.model.llm.device)

        # 拼接 prompt + target
        full_input_ids = torch.cat([inputs.input_ids, target_ids], dim=1)

        # 只对 target 部分计算 loss
        labels = full_input_ids.clone()
        labels[:, :inputs.input_ids.shape[1]] = -100  # 忽略 prompt

        outputs = self.model.llm(
            input_ids=full_input_ids,
            labels=labels,
        )
        return outputs.loss
```

```python
# 来源: 基于快手公开演讲 + RL 业界典型
class LinUCB_Reranker:
    """LinUCB: 在重排层做在线探索
    解决 E&E (Exploit & Explore) 问题
    """
    def __init__(self, num_arms, dim=64):
        self.alpha = 1.5  # 探索系数
        self.A = [np.eye(dim) for _ in range(num_arms)]
        self.b = [np.zeros(dim) for _ in range(num_arms)]

    def select_arm(self, context):
        """context: [dim] 当前上下文特征"""
        scores = []
        for arm in range(len(self.A)):
            A_inv = np.linalg.inv(self.A[arm])
            theta = A_inv @ self.b[arm]
            # UCB: 期望 + 探索项
            ucb = theta @ context + self.alpha * np.sqrt(
                context @ A_inv @ context
            )
            scores.append(ucb)
        return np.argmax(scores)

    def update(self, arm, context, reward):
        """更新参数"""
        self.A[arm] += np.outer(context, context)
        self.b[arm] += reward * context


class BanditExploration:
```

```python
            cand.score,
            avg_dwell / 10.0,
            overlap,
            cand.ctr_pred,
```

## 11. A/B 测试框架

```python
# 来源: 业界典型 A/B 测试 + 快手公开演讲
class ABTestFramework:
    """快手 A/B 测试框架
    - 同时运行多组实验
    - 按 user_id 哈希分桶
    - 实时监控核心指标
    """
    def __init__(self):
        self.experiments = {}
        self.metrics = {}

    def assign_bucket(self, user_id, experiment_id):
        """分配用户到实验桶"""
        exp = self.experiments[experiment_id]
        bucket = hash(f'{user_id}:{experiment_id}') % 100
        if bucket < exp['traffic']:  # 10% 流量
            variant = 'treatment'
        else:
            variant = 'control'
        return variant

    def track_metric(self, user_id, metric_name, value):
        """记录指标"""
        exp_id = self.get_user_experiment(user_id)
        if exp_id:
            self.metrics.setdefault(exp_id, {}).setdefault(
                metric_name, []
            ).append(value)

    def calc_significance(self, exp_id, metric):
        """计算统计显著性"""
        from scipy import stats
        exp_data = self.metrics[exp_id][metric]
        ctrl_data = self.metrics['control'][metric]

        # T 检验
        t_stat, p_value = stats.ttest_ind(exp_data, ctrl_data)
        return {
            'exp_mean': np.mean(exp_data),
            'ctrl_mean': np.mean(ctrl_data),
            'lift': (np.mean(exp_data) - np.mean(ctrl_data)) / np.mean(ctrl_data),
            'p_value': p_value,
            'significant': p_value < 0.05,
        }
```

```
    实时特征回流 → 模型增量训练
```

## 14. 小结：快手推荐核心优势

```
┌────────────────────────────────────────────┐
│         快手 vs 抖音 推荐核心差异           │
├────────────────────────────────────────────┤
│  1. 普惠流量 (基尼系数约束)                │
│  2. 关注关系高权重 (社交属性强)            │
│  3. 完播率权重高 (用户时长导向)             │
│  4. 新人冷启动流量池 (公平曝光)            │
│  5. 同城/发现/关注 三 tab 并重             │
│  6. Pasa LLM Agent 排序 (前沿探索)         │
│  7. 端上重排 (实时反馈)                    │
└────────────────────────────────────────────┘
```

快手推荐系统的关键工程创新：
1. **基尼系数约束**: 保证普惠流量分配，对中小创作者友好
2. **新人冷启动池**: 独立流量池保证新作品有曝光机会
3. **关注关系强化**: 比抖音更重视关注关系，社交粘性更强
4. **长序列建模**: 万级行为序列优化 (SIM 范式)
5. **Pasa LLM 排序**: 用 LLM 做搜索重排的工业级落地
6. **完播率权重**: 鼓励内容创作者做完整、有价值的内容
7. **端上实时重排**: 客户端本地根据实时反馈微调
