### 2.2 召回层：双塔模型 + 多路召回

```python
# 来源: 基于小红书公开演讲 + 业界典型
import torch
import torch.nn as nn


class XHSDoubleTower(nn.Module):
    """小红书双塔召回模型
    特色: 多模态融合 (封面图 + 标题 + 标签)
    """
    def __init__(self, num_notes, num_users, num_tags, dim=128):
        super().__init__()
        # Note 塔
        self.note_tower = NoteTower(num_notes, dim)
        # User 塔
        self.user_tower = XHSUserTower(num_users, num_tags, dim)
        self.temperature = 0.07

    def encode_note(self, note_id, cover_emb, title_emb, tag_embs, author_emb):
        return self.note_tower(note_id, cover_emb, title_emb, tag_embs, author_emb)

    def encode_user(self, user_id, history_ids, tag_pref, profile):
        return self.user_tower(user_id, history_ids, tag_pref, profile)


class NoteTower(nn.Module):
    """笔记塔
    关键: 多模态特征融合 (图像 + 文本 + 标签)
    """
    def __init__(self, num_notes, dim=128):
        super().__init__()
        self.note_emb = nn.Embedding(num_notes, dim)
        self.cover_proj = nn.Linear(512, dim)  # 封面图 embedding (ResNet/ViT)
        self.title_proj = nn.Linear(768, dim)  # 标题文本 embedding (BERT)
        self.tag_proj = nn.Linear(64, dim)  # 标签平均
        self.author_emb = nn.Embedding(5000000, dim)
        self.fusion = nn.Sequential(
            nn.Linear(dim * 5, dim),
            nn.ReLU(),
            nn.Linear(dim, dim),
        )

    def forward(self, note_id, cover_emb, title_emb, tag_embs, author_emb):
        note = self.note_emb(note_id)
        cover = self.cover_proj(cover_emb)
        title = self.title_proj(title_emb)
        tag = self.tag_proj(tag_embs.mean(dim=1))
        author = self.author_emb(author_emb)

        combined = torch.cat([note, cover, title, tag, author], dim=-1)
        return self.fusion(combined)


class XHSUserTower(nn.Module):
    """用户塔
    特色: 标签偏好 + 历史行为 + 用户画像
    """
    def __init__(self, num_users, num_tags, dim=128):
        super().__init__()
        self.user_emb = nn.Embedding(num_users, dim)
        self.tag_pref = nn.Linear(num_tags, dim)
        # 历史行为 Transformer
        self.behavior_attn = nn.MultiheadAttention(dim, num_heads=4, batch_first=True)
        # 用户画像 MLP
        self.profile_mlp = nn.Sequential(
            nn.Linear(64, dim),
            nn.ReLU(),
            nn.Linear(dim, dim),
        )
        self.fusion = nn.Sequential(
            nn.Linear(dim * 3, dim),
            nn.ReLU(),
            nn.Linear(dim, dim),
        )

    def forward(self, user_id, history_ids, tag_pref, profile):
        user = self.user_emb(user_id)
        tag_feat = self.tag_pref(tag_pref)
        # 行为注意力
        if history_ids.shape[1] > 0:
            hist_emb = self.user_emb(history_ids)
            attn_out, _ = self.behavior_attn(hist_emb, hist_emb, hist_emb)
            hist_feat = attn_out.mean(dim=1)
        else:
            hist_feat = torch.zeros_like(user)
        # 画像
        profile_feat = self.profile_mlp(profile)

        combined = torch.cat([user, hist_feat, tag_feat], dim=-1)
        # 还可以加 profile
        combined = torch.cat([combined, profile_feat], dim=-1)
        return self.fusion(combined)
```

```python
class XHSCoverEncoder(nn.Module):
    """小红书封面图编码器
    关键技术:
    - 预训练 ResNet/ViT
    - 提取视觉美学特征
    - 用于相似封面召回
    """
    def __init__(self):
        super().__init__()
        # 视觉编码器 (ViT-B/16 微调)
        from transformers import ViTModel
        self.vit = ViTModel.from_pretrained('google/vit-base-patch16-224')
        # 美学打分头
        self.aesthetic_head = nn.Sequential(
            nn.Linear(768, 256),
            nn.ReLU(),
            nn.Linear(256, 1),
            nn.Sigmoid(),
        )
        # 风格分类头
        self.style_classifier = nn.Linear(768, 20)  # 20 种风格

    def forward(self, image):
        # ViT 提取特征
        outputs = self.vit(image)
        cls_feat = outputs.last_hidden_state[:, 0]  # [B, 768]
        # 美学打分
        aesthetic_score = self.aesthetic_head(cls_feat)
        # 风格
        style_logits = self.style_classifier(cls_feat)
        return cls_feat, aesthetic_score, style_logits

    def encode_for_search(self, image):
        """编码用于检索"""
        feat, _, _ = self.forward(image)
        return nn.functional.normalize(feat, dim=-1)
```

### 3.1 多目标设计

```python
# 来源: 基于小红书技术博客 + 业界典型
class XHSFineRanker(nn.Module):
    """小红书精排模型
    多目标: [CTR, 点赞, 收藏, 评论, 关注, 完播, 主页停留时长]
    """
    def __init__(self, num_notes, num_users, dim=128):
        super().__init__()
        self.note_emb = nn.Embedding(num_notes, dim)
        self.user_emb = nn.Embedding(num_users, dim)
        # DIN 注意力
        self.din_attn = DINAttention(dim)
        # 多模态融合
        self.multimodal_proj = nn.Linear(512, dim)
        # MMoE
        self.mmoe = MMoE(
            num_tasks=7,  # 7 个目标
            num_experts=8,
            input_dim=512,
        )

    def forward(
        self, user_id, history_ids, history_mask,
        candidate_id, cover_emb, title_emb, context,
    ):
        user = self.user_emb(user_id)
        cand = self.note_emb(candidate_id)
        # DIN 兴趣
        hist_emb = self.note_emb(history_ids)
        din_feat = self.din_attn(cand, hist_emb, history_mask)

        # 多模态特征
        mm_feat = self.multimodal_proj(
            torch.cat([cover_emb, title_emb], dim=-1)
        )

        # 拼接
        combined = torch.cat([
            user, din_feat, cand, mm_feat, context
        ], dim=-1)

        # MMoE 多目标
        outputs = self.mmoe(combined)
        return outputs  # [ctr, like, fav, comment, follow, vtr, dwell]
```

```python
# 来源: 基于小红书公开演讲 + 业界典型
class XHSEScore:
    """小红书 E-Score 综合打分公式
    综合多目标分数
    """
    def __init__(self):
        # 权重通过 A/B 测试调优
        self.weights = {
            'ctr': 0.20,
            'like': 0.15,
            'fav': 0.25,  # 收藏权重很高 (小红书特色)
            'comment': 0.10,
            'follow': 0.05,
            'vtr': 0.10,  # 完播率
            'dwell': 0.15,  # 停留时长
        }

    def calc(self, preds):
        score = 0
        for k, w in self.weights.items():
            score += w * preds[k]
        return score
```

### 4.2 Query 理解

```python
# 来源: 基于小红书公开演讲推测
class XHSQueryUnderstanding:
    """小红书 Query 理解
    功能: 改写、NER、意图识别、类目预测
    """
    def __init__(self):
        from transformers import BertForTokenClassification
        # 分词 + NER
        self.ner_model = BertForTokenClassification.from_pretrained(
            'hfl/chinese-roberta-wwm-ext',
            num_labels=20,
        )
        # 意图分类
        self.intent_classifier = BertForSequenceClassification.from_pretrained(
            'hfl/chinese-roberta-wwm-ext',
            num_labels=10,  # 笔记/商品/地点/人/品牌/...
        )
        # 类目预测
        self.category_predictor = BertForSequenceClassification.from_pretrained(
            'hfl/chinese-roberta-wwm-ext',
            num_labels=50,  # 50 个类目
        )
        # 同义词扩展
        self.synonym_dict = self.load_synonyms()

    def understand(self, query):
        # 1. 分词 + NER
        tokens = self.tokenize(query)
        ner_tags = self.ner_model(tokens).logits.argmax(-1)
        entities = self.extract_entities(tokens, ner_tags)

        # 2. 意图分类
        intent_logits = self.intent_classifier(query).logits
        intent = intent_logits.argmax(-1).item()

        # 3. 类目预测
        category_logits = self.category_predictor(query).logits
        category = category_logits.argmax(-1).item()

        # 4. 改写
        rewrites = self.rewrite(query)

        # 5. 同义词扩展
        expansions = self.expand_synonyms(query)

        return {
            'query': query,
            'tokens': tokens,
            'entities': entities,
            'intent': intent,
            'category': category,
            'rewrites': rewrites,
            'expansions': expansions,
        }

    def rewrite(self, query):
        """Query 改写
        - 错别字纠正
        - 口语化 → 书面化
        - 长 query 简化
        """
        return [query]
```

```python
# 来源: 基于小红书公开演讲 + 业界典型
class XHSSearchRanker(nn.Module):
    """小红书搜索精排
    核心: 多模态相关性 (标题 + 封面 + 正文 + OCR + 标签)
    """
    def __init__(self):
        super().__init__()
        from transformers import BertModel
        self.bert = BertModel.from_pretrained('hfl/chinese-roberta-wwm-ext')
        # 视觉编码
        self.vit = ViTModel.from_pretrained('google/vit-base-patch16-224')
        # 多模态融合
        self.fusion = nn.Sequential(
            nn.Linear(768 * 4, 512),
            nn.ReLU(),
            nn.Linear(512, 1),
        )

    def forward(self, query, title, content, cover, tags):
        # 文本相关性
        text_pair = f"[CLS] {query} [SEP] {title} {content}"
        text_feat = self.bert(text_pair).last_hidden_state[:, 0]

        # 视觉相关性
        visual_feat = self.vit(cover).last_hidden_state[:, 0]

        # OCR 文本相关性
        ocr_pair = f"[CLS] {query} [SEP] {self.ocr_text}"
        ocr_feat = self.bert(ocr_pair).last_hidden_state[:, 0]

        # 标签相关性
        tag_pair = f"[CLS] {query} [SEP] {' '.join(tags)}"
        tag_feat = self.bert(tag_pair).last_hidden_state[:, 0]

        # 多模态融合打分
        combined = torch.cat([text_feat, visual_feat, ocr_feat, tag_feat], dim=-1)
        score = torch.sigmoid(self.fusion(combined))
        return score
```

### 4.4 搜索与推荐的融合（**[公开演讲]**）

```python
# 来源: 基于小红书公开演讲 + 业界典型
class SearchRecommendationFusion:
    """小红书的搜索 + 推荐融合 (业界领先)
    - 用户搜索 "显性意图"
    - 用户浏览 "隐性意图"
    - 两者融合生成最终推荐
    """
    def __init__(self):
        self.search_history = RedisZSet('user:search:history')  # 最近搜索
        self.recommend_history = RedisZSet('user:view:history')  # 最近浏览

    def get_user_signals(self, user_id):
        """获取用户显性 + 隐性信号"""
        # 显性: 搜索 query
        search_queries = self.search_history.recent(user_id, top=20)
        # 隐性: 浏览/点赞
        view_notes = self.recommend_history.recent(user_id, top=200)

        # 信号融合
        signals = []
        for q in search_queries:
            signals.append({'type': 'search', 'value': q, 'weight': 1.0})
        for note in view_notes:
            signals.append({'type': 'view', 'value': note, 'weight': 0.3})
        return signals

    def hybrid_recall(self, user_id, num=100):
        """混合召回: 搜索意图 + 浏览兴趣"""
        signals = self.get_user_signals(user_id)

        candidates = {}
        for sig in signals:
            # 每条信号触发召回
            items = self.recall_for_signal(sig, num=num)
            for item in items:
                # 加权累加
                candidates[item.id] = candidates.get(item.id, 0) + sig['weight']

        # 排序
        ranked = sorted(candidates.items(), key=lambda x: -x[1])[:num]
        return ranked
```

```python
# 来源: 基于小红书公开演讲 + 业界典型
class XHSContentModeration:
    """小红书内容审核
    多层审核:
    - 机器审核 (初审)
    - 人工审核 (复审)
    - 用户举报 (兜底)
    """
    def __init__(self):
        # 文本违规检测
        self.text_moderator = BertModerationModel()
        # 图像违规检测
        self.image_moderator = VisionModerationModel()
        # OCR 文字提取
        self.ocr = PaddleOCR()
        # 视频审核
        self.video_moderator = VideoModerationModel()
        # 质量评估
        self.quality_scorer = QualityModel()

    def moderate(self, note):
        """多层审核"""
        # 1. 文本审核
        text_result = self.text_moderator.check(
            note.title + ' ' + note.content
        )
        # 2. OCR + 图像内文字审核
        ocr_text = ''
        image_text_violations = []
        for img in note.images:
            text = self.ocr.recognize(img)
            ocr_text += text
            text_violation = self.text_moderator.check(text)
            if text_violation.is_violation:
                image_text_violations.append(text_violation)

        # 3. 图像审核
        image_results = []
        for img in note.images:
            result = self.image_moderator.check(img)
            image_results.append(result)

        # 4. 视频审核 (如有)
        video_result = None
        if note.video:
            video_result = self.video_moderator.check(note.video)

        # 5. 综合判定
        verdict = self.combine_verdict(
            text_result,
            image_results,
            image_text_violations,
            video_result,
        )
        return verdict

    def combine_verdict(self, text_r, image_rs, text_violations, video_r):
        """综合审核结果"""
        violations = []
        if text_r.is_violation:
            violations.append(text_r)
        for r in image_rs:
            if r.is_violation:
                violations.append(r)
        if text_violations:
            violations.extend(text_violations)
        if video_r and video_r.is_violation:
            violations.append(video_r)

        if not violations:
            return {'action': 'approve', 'violations': []}

        # 严重程度分级
        max_severity = max(v.severity for v in violations)
        if max_severity >= 4:  # 严重 (色情、政治)
            return {'action': 'reject', 'violations': violations}
        elif max_severity >= 2:  # 中等
            return {'action': 'review', 'violations': violations}
        else:  # 轻微
            return {'action': 'approve', 'warnings': violations}
```

### 5.2 违规检测模型（**[业界典型]** + 小红书定制）

```python
# 来源: 基于业界典型 + 小红书公开演讲
class BertModerationModel(nn.Module):
    """BERT 文本违规检测"""
    def __init__(self, num_classes=20):
        super().__init__()
        from transformers import BertModel
        self.bert = BertModel.from_pretrained('hfl/chinese-roberta-wwm-ext')
        self.classifier = nn.Sequential(
            nn.Linear(768, 256),
            nn.ReLU(),
            nn.Dropout(0.1),
            nn.Linear(256, num_classes),
        )
        # 违规等级回归
        self.severity_regressor = nn.Linear(768, 1)

    def forward(self, text):
        outputs = self.bert(text)
        cls_feat = outputs.last_hidden_state[:, 0]
        # 多标签违规分类
        logits = self.classifier(cls_feat)
        # 严重等级
        severity = torch.sigmoid(self.severity_regressor(cls_feat))
        return logits, severity


class VisionModerationModel(nn.Module):
    """视觉违规检测 (黄图/暴力/广告)"""
    def __init__(self):
        super().__init__()
        # EfficientNet 视觉骨干
        self.backbone = timm.create_model('efficientnet_b3', pretrained=True)
        # 违规分类头
        self.classifier = nn.Sequential(
            nn.Linear(1000, 256),
            nn.ReLU(),
            nn.Linear(256, 10),  # 10 类违规
        )

    def forward(self, image):
        feat = self.backbone(image)
        logits = self.classifier(feat)
        return logits
```

```python
class ContentQualityModel:
    """小红书内容质量评估
    评估维度:
    - 原创度 (vs 站内已有内容)
    - 信息量 (长度/词丰富度)
    - 图像质量 (清晰度/构图)
    - 用户互动预测
    """

        self.originality_detector = BertSimilarityModel()
        # 文本质量
        self.text_quality = BertQualityModel()
        # 图像质量
        self.image_quality = ImageAestheticModel()

    def assess(self, note):
        # 1. 原创度: 与站内已有笔记的相似度
        existing = self.search_similar_in_db(note.content)
        max_sim = max((s.score for s in existing), default=0)
        originality = 1.0 - max_sim

        # 2. 信息量
        info_score = self.text_quality.score(note.content)

        # 3. 图像质量
        image_scores = [self.image_quality.score(img) for img in note.images]
        avg_image_score = sum(image_scores) / len(image_scores) if image_scores else 0

        # 4. 综合
        overall = (
            originality * 0.4 +
            info_score * 0.3 +
            avg_image_score * 0.3
        )
        return overall
```

```python
# 来源: 基于小红书公开演讲 + 业界典型
class XHSAntiSpam:
    """小红书风控
    - 笔记刷量检测
    - 评论刷量检测
    - 关注刷量检测
    - 恶意账号识别
    """
    def __init__(self):
        # 行为模式检测
        self.behavior_detector = BehaviorPatternDetector()
        # 设备指纹
        self.device_fingerprint = DeviceFingerprint()
        # 关系图谱
        self.relation_graph = RelationGraph()

    def detect_spam_note(self, note):
        """检测笔记是否为刷量"""
        signals = []

        # 1. 行为模式
        if note.likes > 1000 and note.user_history_avg_likes < 10:
            signals.append(('abnormal_likes', 0.9))

        # 2. 互动账号集中度
        interaction_users = [i.user_id for i in note.interactions]
        concentration = self.calc_concentration(interaction_users)
        if concentration > 0.3:
            signals.append(('high_concentration', concentration))

        # 3. 设备聚集
        devices = [i.device_id for i in note.interactions if i.device_id]
        device_concentration = self.calc_concentration(devices)
        if device_concentration > 0.5:
            signals.append(('device_cluster', device_concentration))

        # 4. 关系图谱分析
        graph_score = self.relation_graph.spam_score(note.user_id)
        if graph_score > 0.7:
            signals.append(('graph_spam', graph_score))

        # 综合判定
        spam_score = sum(w * s for _, s, w in [(k, v, 0.4) for k, v in signals])
        return spam_score > 0.7

    def calc_concentration(self, items):
        """计算集中度 (基尼系数)"""
        from collections import Counter
        counts = Counter(items)
        if len(counts) < 2:
            return 0
        # 简化的基尼系数
        sorted_c = sorted(counts.values())
        n = len(sorted_c)
        cum = sum((i + 1) * v for i, v in enumerate(sorted_c))
        return (2 * cum) / (n * sum(sorted_c)) - (n + 1) / n
```

```python
# 来源: 业界典型 + 小红书公开演讲
class OnDeviceRanking:
    """小红书端上重排
    - 服务端推 50 个候选
    - 端上根据实时行为微调
    """
    def __init__(self):
        # 模型 < 500KB
        self.local_model = TinyDNN(input_dim=64, hidden=32)

    def rerank(self, candidates, recent_actions):
        """recent_actions: 最近 30s 内的浏览/点赞"""
        scores = []
        for cand in candidates:
            # 实时特征
            feature = self.build_realtime_feat(cand, recent_actions)
            score = self.local_model(feature)
            scores.append(score)
        # 重排
        ranked = sorted(zip(candidates, scores), key=lambda x: -x[1])
        return [c for c, s in ranked]

    def build_realtime_feat(self, cand, actions):
        """实时特征构建"""
        # 1. 用户最近停留
        avg_dwell = np.mean([a.dwell for a in actions[-5:]])
        # 2. 用户最近标签
        recent_tags = []
        for a in actions[-3:]:
            if a.tag:
                recent_tags.append(a.tag)
        # 3. 与候选 tag 重合度
        overlap = len(set(cand.tags) & set(recent_tags)) / max(len(cand.tags), 1)
        # 4. 用户最近搜索
        recent_query = actions[-1].query if actions and actions[-1].query else ''
        # 5. query 与候选标题重合度
        query_match = self.calc_text_match(recent_query, cand.title)

        return np.array([
            cand.score,
            avg_dwell / 10.0,
            overlap,
            query_match,
            cand.ctr_pred,
            cand.fav_pred,
        ])
```

```python
# 来源: 基于小红书公开演讲 + 业界典型
class XHSMultimodalModel:
    """小红书多模态内容理解
    - 文本 (标题/正文/OCR)
    - 图像 (封面/内图)
    - 视频 (直播/视频笔记)
    融合提取: 标签/类目/质量/商业价值
    """
    def __init__(self):
        # 视觉编码
        self.vision_encoder = ViTModel.from_pretrained('google/vit-base-patch16-224')
        # 文本编码
        self.text_encoder = BertModel.from_pretrained('hfl/chinese-roberta-wwm-ext')
        # 多模态融合
        self.fusion = nn.Sequential(
            nn.Linear(768 * 2, 512),
            nn.ReLU(),
            nn.Linear(512, 256),
        )
        # 多任务头
        self.tag_head = nn.Linear(256, 100)  # 100 个标签
        self.category_head = nn.Linear(256, 50)  # 50 个类目
        self.quality_head = nn.Linear(256, 1)  # 质量分
        self.commerce_head = nn.Linear(256, 1)  # 商业价值

    def forward(self, image, text):
        # 视觉特征
        visual_feat = self.vision_encoder(image).last_hidden_state[:, 0]
        # 文本特征
        text_feat = self.text_encoder(text).last_hidden_state[:, 0]
        # 融合
        combined = torch.cat([visual_feat, text_feat], dim=-1)
        fused = self.fusion(combined)

        # 多任务
        return {
            'tag_logits': self.tag_head(fused),
            'category_logits': self.category_head(fused),
            'quality_score': torch.sigmoid(self.quality_head(fused)),
            'commerce_score': torch.sigmoid(self.commerce_head(fused)),
        }
```
