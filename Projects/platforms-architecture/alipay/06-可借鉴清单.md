## 6.4 关键代码模板（可直接复用）

### 6.4.1 幂等服务 (Go)

```go
// pkg/idempotent/idempotent.go
// 借鉴蚂蚁 4 层幂等防线
package idempotent

import (
    "context"
    "crypto/sha256"
    "encoding/hex"
    "time"

    "github.com/redis/go-redis/v9"
)

type Service struct {
    rdb *redis.Client
}

func New(rdb *redis.Client) *Service {
    return &Service{rdb: rdb}
}

func (s *Service) Key(scope string, params ...string) string {
    h := sha256.Sum256([]byte(scope + ":" + joinParams(params)))
    return "idem:" + hex.EncodeToString(h[:])
}

// TryLock: 三道防线
func (s *Service) TryLock(ctx context.Context, key string, ttl time.Duration) (bool, error) {
    // L1: Redis SETNX
    ok, err := s.rdb.SetNX(ctx, key, "1", ttl).Result()
    if err != nil {
        return false, err
    }
    return ok, nil
}

func joinParams(parts []string) string {
    out := ""
    for _, p := range parts {
        out += p + "|"
    }
    return out
}
```

# app/services/fingerprint.py
# 借鉴蚁盾 5 维采集
import hashlib
import json
import platform
import uuid
from typing import Dict

class DeviceFingerprint:
    def __init__(self, request_data: Dict):
        self.req = request_data

    def generate(self) -> str:
        # 5 维采集
        raw = {
            "ua": self.req.get("user_agent", ""),
            "ip": self.req.get("ip", ""),
            "screen": self.req.get("screen", ""),
            "tz": self.req.get("timezone", ""),
            "lang": self.req.get("language", ""),
            "fonts": self.req.get("fonts", []),
            "canvas": self.req.get("canvas_hash", ""),
            "webgl": self.req.get("webgl_hash", ""),
        }
        # 哈希编码
        s = json.dumps(raw, sort_keys=True)
        return hashlib.sha256(s.encode()).hexdigest()

### 6.4.3 风控决策 (Python 实时)

```python
# app/services/risk.py
# 借鉴 AlphaRisk 5 道防线
import time
from typing import Dict, Optional

class RiskDecision:
    def __init__(self, rules_engine, ml_model, device_db, graph_db):
        self.rules = rules_engine
        self.model = ml_model
        self.devices = device_db
        self.graph = graph_db

    def decide(self, event: Dict) -> Dict:
        start = time.time()

        # L1: 设备指纹 + 黑名单 (5ms)
        device_id = event.get("device_id")
        if self.devices.is_black(device_id):
            return self._result("REJECT", "device_black", start)

        # L2: 行为异常规则 (15ms)
        rule_risk = self.rules.match(event)
        if rule_risk and rule_risk["level"] == "HIGH":
            return self._result("REJECT", rule_risk["rule"], start)

        # L3: 关系图谱 (20ms)
        graph_risk = self.graph.check_community(event["user_id"])
        if graph_risk["is_fraud_ring"]:
            return self._result("CHALLENGE", "graph_fraud", start)

        # L4: AI 模型评分 (30ms)
        features = self.extract_features(event)
        risk_score = self.model.predict_proba(features)[1]
        if risk_score > 0.7:
            return self._result("REJECT", f"model:{risk_score:.2f}", start)

        # L5: 融合决策
        return self._result("PASS", f"score:{risk_score:.2f}", start)

    def extract_features(self, event):
        # 实际场景: 50+ 维特征
        return [
            event["amount"],
            event["user_age_days"],
            event["device_age_days"],
            event["ip_risk_score"],
            # ... 等等
        ]

    def _result(self, action, reason, start):
        return {
            "action": action,
            "reason": reason,
            "elapsed_ms": int((time.time() - start) * 1000),
        }
```

### 6.4.4 单元化路由 (Go)

```go
# 借鉴蚂蚁单元化寻址
package router

import (
    "hash/crc32"
    "sort"
)

type Cell struct {
    ID       string
    Capacity int
    Health   bool
}

type CellRouter struct {
    cells []Cell
}

func (r *CellRouter) Route(key string) (*Cell, error) {
    // 1. 选健康 cell
    healthy := r.healthyCells()
    if len(healthy) == 0 {
        return nil, ErrNoCell
    }
    // 2. 一致性哈希
    sort.Slice(healthy, func(i, j int) bool {
        return healthy[i].ID < healthy[j].ID
    })
    idx := crc32.ChecksumIEEE([]byte(key)) % uint32(len(healthy))
    return &healthy[idx], nil
}

func (r *CellRouter) healthyCells() []Cell {
    out := []Cell{}
    for _, c := range r.cells {
        if c.Health {
            out = append(out, c)
        }
    }
    return out
}
```
