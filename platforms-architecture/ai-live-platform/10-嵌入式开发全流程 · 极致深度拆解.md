```cpp
// anti_debug.cc
Napi::Value IsDebuggerPresent(const Napi::CallbackInfo& info) {
    #ifdef _WIN32
        return Napi::Boolean::New(info.Env(), 
            IsDebuggerPresent() != FALSE);
    #elif __APPLE__
        // macOS: sysctl 检查
        return Napi::Boolean::New(info.Env(), 
            CheckDebuggerMacOS());
    #else
        // Linux: ptrace
        return Napi::Boolean::New(info.Env(), 
            CheckDebuggerLinux());
    #endif
}

Napi::Value VerifyIntegrity(const Napi::CallbackInfo& info) {
    // 校验自身 .node 文件的 SHA256
    return Napi::Boolean::New(info.Env(), 
        CheckBinaryHash());
}
```
